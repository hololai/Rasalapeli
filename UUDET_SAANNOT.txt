rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    
    // Apufunktiot oikeuksien tarkistamiseen
    function getRole() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role;
    }
    function isSuperAdmin() {
      return request.auth.token.email == 'heikki.laivamaa@gmail.com' || getRole() == 'superadmin';
    }
    function isAdmin() {
      return getRole() == 'admin';
    }

    // Oletuksena kielletään kaikki mitä ei erikseen sallita
    match /{document=**} {
      allow read, write: if false;
    }

    // KARTAN SIJAINNIT: Kaikki kirjautuneet näkevät, vain ylläpito voi muokata
    match /map_locations/{document} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && (isSuperAdmin() || isAdmin());
    }

    // KUVAT: Kaikki kirjautuneet näkevät, vain ylläpito voi luoda/poistaa
    match /images/{imageId} {
      allow read: if request.auth != null;
      allow update: if request.auth != null;
      allow create, delete: if request.auth != null && (isSuperAdmin() || isAdmin());
    }

    // KÄYTTÄJÄT: Omien tietojen lukeminen ja ylläpidon oikeudet
    match /users/{userId} {
      allow create: if request.auth != null && request.auth.uid == userId;
      allow read: if request.auth != null && (request.auth.uid == userId || isSuperAdmin() || isAdmin());
      
      // SUPERADMIN SÄÄNTÖ: saa tehdä mitä vain muiden profiileille
      allow update, delete: if request.auth != null && isSuperAdmin();

      // ADMIN SÄÄNTÖ
      allow update: if request.auth != null && isAdmin() &&
                    (resource.data.role == 'user' || resource.data.role == 'pending') &&
                    (request.resource.data.role == 'user' || request.resource.data.role == 'pending');
                    
      allow delete: if request.auth != null && isAdmin() &&
                    (resource.data.role == 'user' || resource.data.role == 'pending');
    }
  }
}
